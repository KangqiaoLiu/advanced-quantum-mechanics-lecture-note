\section{Nondegenerate Perturbation Theory}
\subsection*{1) First-order correction}
Expand the first-order equation in the $\op H^{(0)}$ representation:
\begin{equation}
 \ket n^{(1)}=\sum_i\ket i^{(0)}{}^{(0)}\!\braket{i}{n^{(1)}},
\end{equation}
\begin{equation}
 \sum_i(\op H^{(0)}-E_n^{(0)})\ket i^{(0)}
 {}^{(0)}\!\braket{i}{n^{(1)}}
 =-(\op H^{(1)}-E_n^{(1)})\ket n^{(0)}.
\end{equation}
Multiply on the left by ${}^{(0)}\!\bra m$ to obtain
\begin{equation}\begin{aligned}
 \sum_i(E_i^{(0)}-E_n^{(0)})\delta_{mi}
 {}^{(0)}\!\braket{i}{n^{(1)}}
 &=-{}^{(0)}\!\mel m{\op H^{(1)}}{n^{(0)}}+E_n^{(1)}\delta_{mn}\\
 &=-H_{mn}^{(1)}+E_n^{(1)}\delta_{mn}.
\end{aligned}\end{equation}
Here $H_{mn}^{(1)}={}^{(0)}\!\mel m{\op H^{(1)}}{n^{(0)}}$ is the matrix element of $\op H^{(1)}$ in the $\op H^{(0)}$ representation. The orthonormality of the eigenstates of $\op H^{(0)}$ has already been used. Thus
\begin{equation}
 (E_m^{(0)}-E_n^{(0)})\,{}^{(0)}\!\braket{m}{n^{(1)}}
 =-H_{mn}^{(1)}+E_n^{(1)}\delta_{mn}.
\end{equation}
For $m=n$,
\begin{equation}
 E_n^{(1)}=H_{nn}^{(1)}.
\end{equation}
For $m\ne n$,
\begin{equation}
 {}^{(0)}\!\braket{m}{n^{(1)}}=
 \frac{H_{mn}^{(1)}}{E_n^{(0)}-E_m^{(0)}}.
\end{equation}
To obtain the full expansion
\begin{equation}
 \ket n^{(1)}=\sum_m\ket m^{(0)}{}^{(0)}\!\braket{m}{n^{(1)}},
\end{equation}
we must also determine ${}^{(0)}\!\braket{n}{n^{(1)}}$.

Normalize to first order, neglecting terms of second and higher order:
\begin{equation}\begin{aligned}
 1=\braket n n
 &=({}^{(0)}\!\bra n+{}^{(1)}\!\bra n)
 (\ket n^{(0)}+\ket n^{(1)})\\
 &\simeq\braket{n^{(0)}}{n^{(0)}}
 +\braket{n^{(1)}}{n^{(0)}}
 +\braket{n^{(0)}}{n^{(1)}}.
\end{aligned}\end{equation}
Since $\braket{n^{(0)}}{n^{(0)}}=1$,
\begin{equation}
 \braket{n^{(0)}}{n^{(1)}}+\braket{n^{(1)}}{n^{(0)}}=0,
\end{equation}
\begin{equation}
 \braket{n^{(0)}}{n^{(1)}}
 +\bigl(\braket{n^{(0)}}{n^{(1)}}\bigr)^*=0.
\end{equation}
The real part of $\braket{n^{(0)}}{n^{(1)}}$ must vanish; it has only an imaginary part. Write
\begin{equation}
 \braket{n^{(0)}}{n^{(1)}}=\ii\alpha.
\end{equation}
Therefore
\begin{equation}\begin{aligned}
 \ket n
 &=\ket n^{(0)}+\sum_m\ket m^{(0)}\braket{m^{(0)}}{n^{(1)}}\\
 &=\ket n^{(0)}+\ii\alpha\ket n^{(0)}
 +\sum_{m\ne n}\frac{H_{mn}^{(1)}}{E_n^{(0)}-E_m^{(0)}}\ket m^{(0)}\\
 &\simeq e^{\ii\alpha}\ket n^{(0)}
 +\sum_{m\ne n}\frac{H_{mn}^{(1)}}{E_n^{(0)}-E_m^{(0)}}\ket m^{(0)}\\
 &\simeq e^{\ii\alpha}\left(
 \ket n^{(0)}+\sum_{m\ne n}\frac{H_{mn}^{(1)}}{E_n^{(0)}-E_m^{(0)}}\ket m^{(0)}\right)
 \quad\text{(to first order)}.
\end{aligned}\end{equation}
The contribution $\braket{n^{(0)}}{n^{(1)}}=\ii\alpha$ to the first-order correction merely supplies a phase factor and does not affect the probability distribution. One may choose $\alpha=0$.

The solution of the stationary Schr\"odinger equation to first order is thus
\begin{equation}
 \boxed{\begin{aligned}
 E_n&=E_n^{(0)}+H_{nn}^{(1)},\\
 \ket n&=\ket n^{(0)}
 +\sum_{m\ne n}\frac{H_{mn}^{(1)}}{E_n^{(0)}-E_m^{(0)}}\ket m^{(0)}.
 \end{aligned}}
\end{equation}
\subsection*{2) Second-order correction}
Expand the second-order equation in the $\op H^{(0)}$ representation:
\begin{equation}
 \ket n^{(2)}=\sum_i\ket i^{(0)}\braket{i^{(0)}}{n^{(2)}},
\end{equation}
\begin{equation}\begin{aligned}
 &\sum_i(\op H^{(0)}-E_n^{(0)})\ket i^{(0)}
 \braket{i^{(0)}}{n^{(2)}}\\
 &\qquad=-\sum_{i\ne n}(\op H^{(1)}-E_n^{(1)})\ket i^{(0)}
 \braket{i^{(0)}}{n^{(1)}}+E_n^{(2)}\ket n^{(0)}.
\end{aligned}\end{equation}
Multiplying by ${}^{(0)}\!\bra m$ gives
\begin{equation}\begin{aligned}
 &\sum_i(E_i^{(0)}-E_n^{(0)})\delta_{mi}
 \braket{i^{(0)}}{n^{(2)}}\\
 &\qquad=-\sum_{i\ne n}(H_{mi}^{(1)}-E_n^{(1)}\delta_{mi})
 \braket{i^{(0)}}{n^{(1)}}+E_n^{(2)}\delta_{mn},\\
 &(E_m^{(0)}-E_n^{(0)})\braket{m^{(0)}}{n^{(2)}}
 =-\sum_{i\ne n}H_{mi}^{(1)}\braket{i^{(0)}}{n^{(1)}}
 +E_n^{(1)}\braket{m^{(0)}}{n^{(1)}}+E_n^{(2)}\delta_{mn}.
\end{aligned}\end{equation}
For $m=n$, taking $\braket{n^{(0)}}{n^{(1)}}=\ii\alpha=0$,
\begin{equation}\begin{aligned}
 E_n^{(2)}
 &=\sum_{i\ne n}H_{ni}^{(1)}\braket{i^{(0)}}{n^{(1)}}
 =\sum_{i\ne n}\frac{H_{ni}^{(1)}H_{in}^{(1)}}{E_n^{(0)}-E_i^{(0)}}\\
 &=\sum_{i\ne n}\frac{|H_{in}^{(1)}|^2}{E_n^{(0)}-E_i^{(0)}}.
\end{aligned}\end{equation}
For $m\ne n$, the coefficient $\braket{m^{(0)}}{n^{(2)}}$ can also be found (see the textbook and calculate it yourself).

The approximate solution of the stationary Schr\"odinger equation through second order is
\begin{equation}
 \boxed{\begin{aligned}
 E_n&=E_n^{(0)}+H_{nn}^{(1)}
 +\sum_{i\ne n}\frac{|H_{in}^{(1)}|^2}{E_n^{(0)}-E_i^{(0)}},\\
 \ket n&=\ket n^{(0)}
 +\sum_{i\ne n}\frac{H_{in}^{(1)}}{E_n^{(0)}-E_i^{(0)}}\ket i^{(0)}
 +\ket n^{(2)}.
 \end{aligned}}
\end{equation}
\subsection*{3) Discussion of convergence}
The condition for perturbation theory to be applicable is convergence, namely,
\begin{equation}
 \left|\frac{H_{in}^{(1)}}{E_n^{(0)}-E_i^{(0)}}\right|\ll1,
 \quad\text{i.e.,}\quad
 |H_{in}^{(1)}|\ll|E_n^{(0)}-E_i^{(0)}|.
\end{equation}
This is the meaning of $\op H^{(1)}$ being much smaller than $\op H^{(0)}$: in the $H^{(0)}$ representation, its matrix elements are much smaller than the differences between the corresponding energy levels.

It follows that:
\begin{enumerate}
\item The above method applies only when $\op H^{(0)}$ has a discrete spectrum. For a continuous spectrum, $|E_n^{(0)}-E_i^{(0)}|\to0$ and the above inequality cannot hold.
\item It cannot be used when $\op H^{(0)}$ has degeneracies. In that case, $|E_n^{(0)}-E_i^{(0)}|=0$ and the inequality cannot hold either.
\end{enumerate}
\textbf{Example: A charged harmonic oscillator in an external electric field.}
Let the electric charge be $q$ and the electric-field strength be $\mathcal E$. The Hamiltonian is
\begin{equation}
 \op H=-\frac{\hbar^2}{2m}\frac{\dd^2}{\dd x^2}
 +\frac12m\omega^2x^2-q\mathcal E x.
\end{equation}
For a weak electric field $\mathcal E$, take
\begin{equation}
 \op H=\op H^{(0)}+\op H^{(1)},\quad
 \op H^{(0)}=-\frac{\hbar^2}{2m}\frac{\dd^2}{\dd x^2}+\frac12m\omega^2x^2,
 \quad\op H^{(1)}=-q\mathcal E x.
\end{equation}
Zeroth-order approximation:
\begin{equation}
 \op H^{(0)}\ket n^{(0)}=E_n^{(0)}\ket n^{(0)},\quad
 E_n^{(0)}=\left(n+\frac12\right)\hbar\omega,
 \quad\braket{x^{(0)}}{n^{(0)}}=\psi_n^{(0)}(x).
\end{equation}
The harmonic-oscillator eigenfunctions are Hermite-polynomial functions, with discrete, nondegenerate energies and separated variables.

First-order correction:
\begin{equation}
 E_n^{(1)}=H_{nn}^{(1)}=-q\mathcal E x_{nn},
\end{equation}
\begin{equation}
 x_{nm}={}^{(0)}\!\mel n x{m^{(0)}}
 =\sqrt{\frac{\hbar}{m\omega}}
 \left(\sqrt{\frac{n+1}{2}}\delta_{n+1,m}
 +\sqrt{\frac n2}\delta_{n-1,m}\right),
 \quad x_{nn}=0,\ E_n^{(1)}=0.
\end{equation}
For the first-order change of the state,
\begin{equation}\begin{aligned}
 \ket n^{(1)}
 &=\sum_{m\ne n}\frac{H_{mn}^{(1)}}{E_n^{(0)}-E_m^{(0)}}\ket m^{(0)}
 =-q\mathcal E\sum_{m\ne n}\frac{x_{mn}}{E_n^{(0)}-E_m^{(0)}}\ket m^{(0)}\\
 &=-q\mathcal E\sqrt{\frac{\hbar}{2m\omega}}
 \left(\frac{\sqrt{n+1}}{E_n^{(0)}-E_{n+1}^{(0)}}\ket{n+1}^{(0)}
 +\frac{\sqrt n}{E_n^{(0)}-E_{n-1}^{(0)}}\ket{n-1}^{(0)}\right)\\
 &=\frac{q\mathcal E}{\sqrt{2m\hbar\omega^3}}
 \left(\sqrt{n+1}\ket{n+1}^{(0)}-\sqrt n\ket{n-1}^{(0)}\right).
\end{aligned}\end{equation}
In the coordinate representation,
\begin{equation}
 \psi_n^{(1)}(x)=\braket x{n^{(1)}}
 =\frac{q\mathcal E}{\sqrt{2m\hbar\omega^3}}
 \left[\sqrt{n+1}\,\psi_{n+1}^{(0)}(x)
 -\sqrt n\,\psi_{n-1}^{(0)}(x)\right].
\end{equation}
Second-order correction:
\begin{equation}\begin{aligned}
 E_n^{(2)}&=\sum_{m\ne n}\frac{|H_{mn}^{(1)}|^2}{E_n^{(0)}-E_m^{(0)}}\\
 &=\frac{q^2\mathcal E^2\hbar}{m\omega}
 \sum_{m\ne n}\frac{\bigl|\sqrt{(n+1)/2}\,\delta_{m,n+1}
 +\sqrt{n/2}\,\delta_{m,n-1}\bigr|^2}{E_n^{(0)}-E_m^{(0)}}\\
 &=-\frac{q^2\mathcal E^2}{2m\omega^2}.
\end{aligned}\end{equation}
Thus, keeping the first two nonzero corrections,
\begin{equation}
 \boxed{\begin{aligned}
 E_n&=\left(n+\frac12\right)\hbar\omega
 -\frac{q^2\mathcal E^2}{2m\omega^2},\\
 \psi_n(x)&\simeq\psi_n^{(0)}(x)
 +\frac{q\mathcal E}{\sqrt{2m\hbar\omega^3}}
 \left(\sqrt{n+1}\,\psi_{n+1}^{(0)}(x)
 -\sqrt n\,\psi_{n-1}^{(0)}(x)\right).
 \end{aligned}}
\end{equation}
Can an exact solution be found? Start from
\begin{equation}
 \left(-\frac{\hbar^2}{2m}\frac{\dd^2}{\dd x^2}
 +\frac12m\omega^2x^2-q\mathcal E x\right)\psi(x)=E\psi(x).
\end{equation}
Complete the square:
\begin{equation}
 \left[-\frac{\hbar^2}{2m}\frac{\dd^2}{\dd x^2}
 +\frac12m\omega^2\left(x-\frac{q\mathcal E}{m\omega^2}\right)^2\right]\psi(x)
 =\left(E+\frac{q^2\mathcal E^2}{2m\omega^2}\right)\psi(x).
\end{equation}
Put $\xi=x-q\mathcal E/(m\omega^2)$; then
\begin{equation}
 \left(-\frac{\hbar^2}{2m}\frac{\dd^2}{\dd\xi^2}
 +\frac12m\omega^2\xi^2\right)\psi(\xi)
 =\left(E+\frac{q^2\mathcal E^2}{2m\omega^2}\right)\psi(\xi).
\end{equation}
The solution is
\begin{equation}
 \boxed{\begin{aligned}
 E_n&=\left(n+\frac12\right)\hbar\omega
 -\frac{q^2\mathcal E^2}{2m\omega^2},\\
 \psi_n(\xi)&=\psi_n^{(0)}\left(x-\frac{q\mathcal E}{m\omega^2}\right),
 \end{aligned}}
\end{equation}
where $\psi_n^{(0)}$ is a Hermite-polynomial function.

The energy computed through second order is already exact.
