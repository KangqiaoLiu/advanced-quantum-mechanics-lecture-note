\subsection*{7) Proportionality constants for particle-number operators}
Since
\begin{align*}
n_i
&=\mel{0,\ldots,n_i,\ldots}{\op n_i}{0,\ldots,n_i,\ldots}\\
&=\mel{0,\ldots,n_i,\ldots}{\op a_i^\dagger\op a_i}{0,\ldots,n_i,\ldots},
\end{align*}
we have, noting that $n_1=n_2=\cdots=n_{i-1}=0$ in the following relations,
\begin{align*}
\op a_i\ket{0,\ldots,n_i,\ldots}
&=\sqrt{n_i}\ket{0,\ldots,n_i-1,\ldots},\\
\op a_i^\dagger\ket{0,\ldots,n_i-1,\ldots}
&=\sqrt{n_i}\ket{0,\ldots,n_i,\ldots}.
\end{align*}
The state can be written
\[
\ket{n_1,\ldots,n_i,\ldots}
=A(n_1,\ldots,n_{i-1},\ldots)
(\op a_1^\dagger)^{n_1}(\op a_2^\dagger)^{n_2}\cdots
(\op a_{i-1}^\dagger)^{n_{i-1}}
\ket{0,\ldots,n_i,\ldots}.
\]
Pay attention to the order of creation operators. When $\op a_j^\dagger$ acts, $n_1=n_2=\cdots=n_{j-1}=0$. For bosons, the commutation relations give
\begin{align*}
\op a_i\ket{n_1,\ldots,n_i,\ldots}
&=A(n_1,\ldots,n_{i-1},\ldots)
(\op a_1^\dagger)^{n_1}\cdots
(\op a_{i-1}^\dagger)^{n_{i-1}}
\op a_i\ket{0,\ldots,n_i,\ldots}\\
&=A(n_1,\ldots,n_{i-1},\ldots)
(\op a_1^\dagger)^{n_1}\cdots
(\op a_{i-1}^\dagger)^{n_{i-1}}
\sqrt{n_i}\ket{0,\ldots,n_i-1,\ldots}\\
&=\sqrt{n_i}\ket{n_1,\ldots,n_i-1,\ldots}.
\end{align*}
Likewise,
\[\op a_i^\dagger\ket{n_1,\ldots,n_i,\ldots}
=\sqrt{n_i+1}\ket{n_1,\ldots,n_i+1,\ldots}.\]
For fermions, the anticommutation relations give
\begin{align*}
\op a_i\ket{n_1,\ldots,n_i,\ldots}
&=A(n_1,\ldots,n_{i-1},\ldots)
(\op a_1^\dagger)^{n_1}\cdots
(\op a_{i-1}^\dagger)^{n_{i-1}}
(-1)^{\sum_{j=1}^{i-1}n_j}
\op a_i\ket{0,\ldots,n_i,\ldots}\\
&=(-1)^{\sum_{j=1}^{i-1}n_j}\sqrt{n_i}
\ket{n_1,\ldots,n_i-1,\ldots},
\end{align*}
and
\[\op a_i^\dagger\ket{n_1,\ldots,n_i,\ldots}
=(-1)^{\sum_{j=1}^{i-1}n_j}\sqrt{n_i+1}
\ket{n_1,\ldots,n_i+1,\ldots}.\]
Hence for either bosons or fermions,
\begin{align*}
\mel{n_1,\ldots,n_i,\ldots}{\op n_i}{n_1,\ldots,n_i,\ldots}
&=\mel{n_1,\ldots,n_i,\ldots}{\op a_i^\dagger\op a_i}{n_1,\ldots,n_i,\ldots}\\
&=n_i.
\end{align*}
